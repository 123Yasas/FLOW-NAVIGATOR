ESP32 Arduino C++ Firmware Code Generator (Production Hardware Ready)

Flash this code directly to your ESP32 via Arduino IDE or PlatformIO. It connects to Wi-Fi, initializes dual VL53L0X ToF sensors on I2C, tracks pedestrian entrance vs exit vectors, and posts HTTP JSON counts to /api/sensors/count. 

// =========================================================================
// FlowNavigator ESP32-S3 Dual VL53L0X ToF Counter (Sona College Ground)
// Target: ESP32-S3 | Dual TwoWire(0) & TwoWire(1) | Endpoint: /api/crowd
// =========================================================================
#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <VL53L0X.h>

const char* ssid = "SVVT AirFiber";
const char* password = "YOUR_WIFI_PASSWORD";
const char* SERVER_URL = "http://192.168.31.10:5000/api/crowd";

TwoWire I2C_Entry = TwoWire(0);
TwoWire I2C_Exit = TwoWire(1);
VL53L0X entrySensor;
VL53L0X exitSensor;

int startingPeople = 0;
int crowd = startingPeople;
int entryCount = 0;
int exitCount = 0;
const int CAPACITY = 10;
const int VENUE_B_LIMIT = 8;
const int DETECTION_LIMIT_MM = 50;

bool entryDetected = false;
bool exitDetected = false;
String lastEvent = "STARTED";
unsigned long lastUpload = 0;
const unsigned long UPLOAD_INTERVAL = 2000;

void setup() {
  Serial.begin(115200);
  delay(2000);
  Serial.println("FlowNavigator Starting");
  I2C_Entry.begin(8, 9);
  I2C_Entry.setClock(100000);
  I2C_Exit.begin(10, 11);
  I2C_Exit.setClock(100000);
  entrySensor.setBus(&I2C_Entry);
  exitSensor.setBus(&I2C_Exit);
  entrySensor.setTimeout(1000);
  exitSensor.setTimeout(1000);
  if (!entrySensor.init()) { Serial.println("Entry sensor failed"); while (1) { delay(1000); } }
  if (!exitSensor.init()) { Serial.println("Exit sensor failed"); while (1) { delay(1000); } }
  entrySensor.setMeasurementTimingBudget(50000);
  exitSensor.setMeasurementTimingBudget(50000);
  entrySensor.startContinuous(100);
  exitSensor.startContinuous(100);
  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nWiFi Connected! ESP32 IP: " + WiFi.localIP().toString());
}

void loop() {
  uint16_t entryDistance = entrySensor.readRangeContinuousMillimeters();
  bool entryTimeout = entrySensor.timeoutOccurred();
  uint16_t exitDistance = exitSensor.readRangeContinuousMillimeters();
  bool exitTimeout = exitSensor.timeoutOccurred();

  if (!entryTimeout && entryDistance != 65535) {
    if (entryDistance < DETECTION_LIMIT_MM && !entryDetected) {
      entryDetected = true;
      if (crowd < CAPACITY) { crowd++; entryCount++; lastEvent = "PERSON_ENTERED"; }
      else { lastEvent = "ENTRY_BLOCKED"; }
    }
    if (entryDistance >= DETECTION_LIMIT_MM) { entryDetected = false; }
  }

  if (!exitTimeout && exitDistance != 65535) {
    if (exitDistance < DETECTION_LIMIT_MM && !exitDetected) {
      exitDetected = true;
      if (crowd > 0) { crowd--; exitCount++; lastEvent = "PERSON_EXITED"; }
      else { lastEvent = "EXIT_IGNORED"; }
    }
    if (exitDistance >= DETECTION_LIMIT_MM) { exitDetected = false; }
  }

  String status = (crowd >= CAPACITY) ? "NO_ENTRY_ALLOWED" : (crowd >= VENUE_B_LIMIT ? "RECOMMEND_VENUE_B" : "ENTRY_ALLOWED");

  if (millis() - lastUpload >= UPLOAD_INTERVAL) {
    lastUpload = millis();
    if (WiFi.status() == WL_CONNECTED) {
      HTTPClient http;
      if (http.begin(SERVER_URL)) {
        http.setTimeout(1500);
        http.addHeader("Content-Type", "application/json");
        float entryCM = (!entryTimeout && entryDistance != 65535) ? entryDistance / 10.0 : -1.0;
        float exitCM = (!exitTimeout && exitDistance != 65535) ? exitDistance / 10.0 : -1.0;
        String payload = "{\"sensor_id\":\"ESP32-S3-VL53L0X-GATE\",\"zone_id\":\"zone-a-sona-college\",\"venue_id\":\"sona-college\",\"entryCount\":" + String(entryCount) + ",\"exitCount\":" + String(exitCount) + ",\"currentCrowd\":" + String(crowd) + ",\"capacity\":" + String(CAPACITY) + ",\"entryDistanceCm\":" + String(entryCM, 1) + ",\"exitDistanceCm\":" + String(exitCM, 1) + ",\"status\":\"" + status + "\",\"lastEvent\":\"" + lastEvent + "\"}";
        int responseCode = http.POST(payload);
        Serial.print("Backend HTTP: "); Serial.println(responseCode);
        http.end();
      }
    }
  }
  delay(100);
}
